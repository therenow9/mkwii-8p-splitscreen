# RR's settings in a 2-player game, the 1-4P reference for 6p-royale-settings (M4 D3): 2 Players, the multiplayer
# menu's Settings button to RR's page select (a shot), Race -> Menu -> Sound -> Down (a shot: Knockout), back to Race,
# then Battle -> Royale and its page (a shot), Up from Save (a shot), Back, Battle's page (a shot) and Back twice:
# every button and row as RR draws them. GC ports 1-2.

wait 175

wait 40
press 1 a
wait 300
press 2 a        # controller registration
wait 150
press 1 a        # OK
wait 480
press 1 up       # multiplayer menu: VS -> the Settings button (top right)
wait 120
press 1 a        # Settings
wait 600         # page select, held for its shot
press 1 right    # Race -> Menu
wait 60
press 1 right    # -> Sound
wait 60
press 1 down     # -> Knockout, or Misc where Knockout is greyed
wait 300         # held for its shot
press 1 up       # back to Sound from either
wait 60
press 1 left     # -> Menu
wait 60
press 1 left     # -> Race
wait 60
press 1 down     # -> Battle
wait 60
press 1 down     # -> Royale
wait 60
press 1 a        # the Royale page
wait 600         # held for its shot
press 1 up       # the row above Save: Starting Lives, or All Items Can Land where the Battle Royale rows are greyed
wait 300         # held for its shot
press 1 b        # back to page select
wait 300
press 1 down     # page select starts on Race again: -> Battle
wait 60
press 1 a        # the Battle page: no row greyed after Royale's
wait 300         # held for its shot
press 1 b        # back to page select
